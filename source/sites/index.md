<script>
(function () {
  const root = document.querySelector('[data-sites-root]');
  if (!root) return;
  const input = root.querySelector('[data-sites-search]');
  const chips = Array.from(root.querySelectorAll('[data-sites-filter]'));
  const cards = Array.from(root.querySelectorAll('[data-sites-card]'));
  const empty = root.querySelector('[data-sites-empty]');
  let active = 'all';
  function update() {
    const keyword = (input && input.value ? input.value : '').trim().toLowerCase();
    let visibleCount = 0;
    cards.forEach(function (card) {
      const matchCategory = active === 'all' || card.dataset.category === active;
      const matchSearch = !keyword || (card.dataset.search || '').toLowerCase().includes(keyword);
      const show = matchCategory && matchSearch;
      card.hidden = !show;
      if (show) visibleCount += 1;
    });
    if (empty) empty.hidden = visibleCount !== 0;
  }
  chips.forEach(function (chip) {
    chip.addEventListener('click', function () {
      active = chip.dataset.sitesFilter || 'all';
      chips.forEach(function (item) { item.classList.toggle('is-active', item === chip); });
      update();
    });
  });
  if (input) input.addEventListener('input', update);
}());
</script>
